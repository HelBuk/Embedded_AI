    \section{Metrcis}
    \begin{frame}{Detection Metrics}
        \vspace{0.6\baselineskip}
        For a predicted box \(p\) and ground-truth box \(g\): \\
        \small
          \[
            \IoU(p,g) = \frac{|p\cap g|}{|p\cup g|},
            \Precision = \frac{\TP}{\TP+\FP},
            \Recall = \frac{\TP}{\TP+\FN},
            \AP_{\tau} = \int_{0}^{1} p(r;\tau)\,dr,
          \]
          \[
            \mAP_{0.50} = \frac{1}{C}\sum_{c=1}^{C}\AP_{c,\,0.50},
          \]
          \[
            \mAP_{[0.50:0.95]} = \frac{1}{C}\sum_{c=1}^{C}\frac{1}{10}
                                 \sum_{k=0}^{9}\AP_{c,\,(0.50+0.05k)}.
          \]
        \begin{itemize}\itemsep2pt
            \item \(\tau\): IoU threshold for TP/FP (0.50 for mAP50; 0.50–0.95 for COCO mAP).
            \item AP is area under the precision–recall curve (confidence threshold is swept).
            \item C is the number of object classes we’re evaluating over.
        \end{itemize}
    \end{frame}
